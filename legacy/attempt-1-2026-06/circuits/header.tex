% Shared preamble for all CircuiTikZ schematics in this project.
% Each circuit file should be a self-contained `standalone` document that
% \input{}s this file.  Compile with:  pdflatex foo.tex
\documentclass[border=6pt]{standalone}
\usepackage{circuitikz}
\usepackage{amsmath}
\usepackage{xcolor}
\usetikzlibrary{calc,positioning}

% Project style — tuned for analog CMOS schematics
\ctikzset{
    bipoles/length=1.1cm,
    transistors/arrow pos=end,
    nodes width=0.06,
    logic ports=ieee,
}

% Default tikz options applied inside circuitikz environments
\tikzset{
    every picture/.style={
        line cap=round,
        line join=round,
        line width=0.6pt,
        font=\small,
    },
    every node/.style={font=\small},
}

% Convenience macros
\newcommand{\Vdd}{\ensuremath{V_{dd}}}
\newcommand{\Vddb}{\ensuremath{V_{ddb}}}
\newcommand{\Vss}{\ensuremath{V_{ss}}}

% --- Annotation mode: voltages (D2) vs branch currents (D3) ---------------
% A wrapper sets \showItrue for the current sheet; default is the voltage sheet.
% \vlab{...} renders only on the voltage sheet, \ilab{...} only on the current
% sheet, so one shared *_fig.tex drawing produces both D2 and D3 figures.
\newif\ifshowI \showIfalse
\newcommand{\vlab}[1]{\ifshowI\else#1\fi}
\newcommand{\ilab}[1]{\ifshowI#1\fi}
% Styled node-voltage tag (blue) and branch-current tag (red).
\newcommand{\Vtag}[1]{\vlab{\textcolor{blue!60!black}{\footnotesize $#1$}}}
\newcommand{\Itag}[1]{\ilab{\textcolor{red!65!black}{\footnotesize $#1$}}}
